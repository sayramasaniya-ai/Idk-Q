local function nowMs()

local function asNumber(v, fallback)
    local n = tonumber(v)
    if n == nil then
        return fallback or 0
    end
    return n
end

local function safeLower(v)

end

local function compactNumber(n)
    n = asNumber(n, 0)
    local absN = math.abs(n)

    if absN >= 1e12 then
        return string.format("%.1fT", n / 1e12):gsub("%.0T", "T")
    elseif absN >= 1e9 then
        return string.format("%.1fB", n / 1e9):gsub("%.0B", "B")
    elseif absN >= 1e6 then
        return string.format("%.1fM", n / 1e6):gsub("%.0M", "M")
    elseif absN >= 1e3 then
        return string.format("%.1fK", n / 1e3):gsub("%.0K", "K")
    end

    return tostring(math.floor(n))
end

local function getGenText(trade)
    if trade and trade.genText and tostring(trade.genText) ~= "" then
        return tostring(trade.genText)
    end

    local gv = asNumber(trade and trade.genValue, 0)
    return "$" .. compactNumber(gv) .. "/s"
end

local function doHttpGet(url)
    if not request then
        return false, "No request function"
    end

    local ok, res = pcall(function()
        return request({
            Url = url,
            Method = "GET",
            Headers = {
                ["Accept"] = "application/json"
            }
        })
    end)

    if not ok or not res then
        return false, "Request failed"
    end

    local body = res.Body or res.body
    if type(body) ~= "string" then
        return false, "Invalid body"
    end

    return true, body
end

        return false
    end

    if CONFIG.FILTER_EXPIRED and isExpiredTrade(trade) then
        return false
    end

    if not isValidStatus(trade) then
        return false
    end

    local mutation = tostring(trade.mutation or "None")
    local hasMutation = mutation ~= "" and mutation ~= "None"

    if onlyMutated and not hasMutation then
        return false
    end

    if searchText ~= "" then
        local haystack = table.concat({
            safeLower(trade.animalDisplayName),
            safeLower(trade.animal),
            safeLower(trade.username),
            safeLower(trade.traits),
            safeLower(trade.rarity),
            safeLower(trade.mutation),
        }, " | ")

        if not haystack:find(searchText, 1, true) then
            return false
        end
    end

    return true
end

local function renderTrades()
    clearTradeList()

    local searchText = safeLower(SearchBox.Text)
    local visible = {}
    for _, trade in ipairs(allTrades) do
        if tradeMatchesFilters(trade, searchText) then
            table.insert(visible, trade)
        end
    end

    table.sort(visible, function(a, b)
        return asNumber(a.genValue, 0) > asNumber(b.genValue, 0)
    end)

    for i, trade in ipairs(visible) do
        createTradeCard(trade, i)
    end

    FooterLabel.Text = string.format(
        "%d trade%s encontrado%s",
        #visible,
        #visible == 1 and "" or "s",
        #visible == 1 and "" or "s"
    )

    return #visible
end

-- ================================================
--  FETCH
-- ================================================
local isFetching = false
local pendingFetch = false
local fetchNonce = 0

local function fetchTrades()
    if isFetching then
        pendingFetch = true
        return
    end

    isFetching = true
    fetchNonce += 1
    local myNonce = fetchNonce

    StatusLabel.Text = "⏳ Actualizando..."

    local ok, raw = doHttpGet(CONFIG.API_URL)
    if not ok then
        StatusLabel.Text = "❌ Error HTTP: " .. tostring(raw):sub(1, 80)
        isFetching = false
        if pendingFetch then
            pendingFetch = false
            task.defer(fetchTrades)
        end
        return
    end

    local decoded = decodeTrades(raw)
    if type(decoded) ~= "table" then
        allTrades = {}
        StatusLabel.Text = "❌ JSON inválido"
        isFetching = false
        if pendingFetch then
            pendingFetch = false
            task.defer(fetchTrades)
        end
        return
    end

    allTrades = decoded
    table.sort(allTrades, function(a, b)
        return asNumber(a.genValue, 0) > asNumber(b.genValue, 0)
    end)

    local visibleCount = renderTrades()
    if myNonce == fetchNonce then
        StatusLabel.Text = string.format("✅ %d trades cargados · %d visibles", #allTrades, visibleCount)
    end

    isFetching = false

    if pendingFetch then
        pendingFetch = false
        task.defer(fetchTrades)
    end
end
